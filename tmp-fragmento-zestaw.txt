if (!('IntersectionObserver' in window) || reduce) {
  revealAll();
} else {
  const io = new IntersectionObserver(entries => {
    entries.forEach(e => { if (e.isIntersecting) { e.target.classList.add('zp-visible'); io.unobserve(e.target); } });
  }, { threshold:.12, rootMargin:'0px 0px -50px 0px' });
  document.querySelectorAll('.zp-reveal').forEach(el => io.observe(el));
  
}

/* numeros que sobem */
(function(){
  const nums = [...document.querySelectorAll('[data-count]')];
  const setFinal = el => { el.textContent = (el.dataset.prefix||'') + el.dataset.count + (el.dataset.suffix||''); };
  const runCount = el => {
    if (el.dataset.done) return;
    el.dataset.done = '1';
    if (reduce) { setFinal(el); return; }
    const target = +el.dataset.count, pre = el.dataset.prefix||'', suf = el.dataset.suffix||'';
    const dur = 1300, t0 = performance.now();
    const tick = t => {
      const p = Math.min((t - t0)/dur, 1);
      el.textContent = pre + Math.round(target * (1 - Math.pow(1-p,3))) + suf;
      if (p < 1) requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  };
  if (!('IntersectionObserver' in window) || reduce) {
    nums.forEach(setFinal);
  } else {
    const counters = new IntersectionObserver(entries => {
      entries.forEach(e => { if (e.isIntersecting) { counters.unobserve(e.target); runCount(e.target); } });
    }, { threshold:.6 });
    nums.forEach(el => counters.observe(el));
    setTimeout(() => nums.forEach(el => { if (!el.dataset.done) setFinal(el); }), 3000);
  }
})();

/* carrossel da galeria */
(function(){
  const trilho = document.getElementById('zp-carrossel');
  if (!trilho) return;
  const irPara = destino => {
    const inicio = trilho.scrollLeft, delta = destino - inicio, t0 = performance.now(), dur = 420;
    const passo = t => {
      const p = Math.min((t - t0)/dur, 1);
      trilho.scrollLeft = inicio + delta * (1 - Math.pow(1-p, 3));
      if (p < 1) requestAnimationFrame(passo);
    };
    requestAnimationFrame(passo);
    setTimeout(() => { trilho.scrollLeft = destino; }, dur + 80);
  };
  const item = trilho.querySelector('.zp-gal-item');
  const salto = () => (item ? item.offsetWidth : 240) + 16;
  document.getElementById('zp-galNext').addEventListener('click', () => irPara(trilho.scrollLeft + salto()));
  document.getElementById('zp-galPrev').addEventListener('click', () => irPara(trilho.scrollLeft - salto()));
})();
