(function () {

    var pageNav = Array.prototype.slice.call(document.querySelectorAll('.sat-pages-nav'));
    var pageActive = Math.min(2, pageCards.length - 1);
    var pagePositions = ['pos-active', 'pos-left-1', 'pos-right-1', 'pos-left-2', 'pos-right-2', 'pos-hidden'];
    var pageTimer = null;
    var pagePaused = false;
    var pageStartX = null;
    var pageDragging = false;
    var pageSuppressClick = false;
    var reducePageMotion = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    function renderPages() {
      var total = pageCards.length;
      pageCards.forEach(function (card, idx) {
        var diff = idx - pageActive;
        if (diff > total / 2) diff -= total;
        if (diff < -total / 2) diff += total;
        pagePositions.forEach(function (cls) { card.classList.remove(cls); });
        if (diff === 0) card.classList.add('pos-active');
        else if (diff === 1) card.classList.add('pos-right-1');
        else if (diff === 2) card.classList.add('pos-right-2');
        else if (diff === -1) card.classList.add('pos-left-1');
        else if (diff === -2) card.classList.add('pos-left-2');
        else card.classList.add('pos-hidden');
        card.dataset.idx = idx;
        card.setAttribute('aria-hidden', diff === 0 ? 'false' : 'true');
      });
    }
    function movePages(delta) {
      pageActive = (pageActive + delta + pageCards.length) % pageCards.length;
      renderPages();
    }
    function movePagesTo(idx) {
      pageActive = (idx + pageCards.length) % pageCards.length;
      renderPages();
    }
    function stopPageAutoplay() {
      if (pageTimer) { clearInterval(pageTimer); pageTimer = null; }
    }
    function startPageAutoplay() {
      stopPageAutoplay();
      if (reducePageMotion) return;
      pageTimer = setInterval(function () {
        if (!pagePaused && !document.hidden) movePages(1);
      }, 1800);
    }
    function resetPageAutoplay() {
      stopPageAutoplay();
      if (!reducePageMotion) setTimeout(startPageAutoplay, 6000);
    }
    function startPageDrag(x) {
      pageStartX = x;
      pageDragging = true;
      pagePaused = true;
    }
    function endPageDrag(x) {
      if (!pageDragging || pageStartX === null) { pagePaused = false; return; }
      var dx = x - pageStartX;
      if (Math.abs(dx) > 40) {
        pageSuppressClick = true;
        setTimeout(function () { pageSuppressClick = false; }, 250);
        movePages(dx < 0 ? 1 : -1);
        resetPageAutoplay();
      }
      pageStartX = null;
      pageDragging = false;
      pagePaused = false;
    }

    pageCards.forEach(function (card, idx) {
      card.addEventListener('click', function () {
        if (pageSuppressClick) return;
        if (idx !== pageActive) { movePagesTo(idx); resetPageAutoplay(); }
      });
    });
    pageNav.forEach(function (button) {
      button.addEventListener('click', function () {
        movePages(parseInt(button.getAttribute('data-dir'), 10) || 1);
        resetPageAutoplay();
      });

    document.addEventListener('pointerup', function (e) { endPageDrag(e.clientX); });
