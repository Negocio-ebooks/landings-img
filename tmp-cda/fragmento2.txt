<script>
(function () {
  'use strict';
  var root = document.getElementById('cda-page');
  if (!root) return;
  var original = root.querySelector('#cda-original');
  var savings = root.querySelector('#cda-savings');
  var price = root.querySelector('#cda-current-price');
  function resolved(t) { return !!t && t.indexOf('{{') === -1; }
  function refreshPrices() {
    try {
      var s = savings.textContent.trim(), o = original.textContent.trim(), p = price.textContent.trim();
      var show = resolved(s) && /[1-9]/.test(s) && !/[-−]/.test(s) && resolved(o) && resolved(p) && o !== p;
      root.querySelector('#cda-old-row').hidden = !show;
    } catch (e) {}
  }
  refreshPrices();
  if ('MutationObserver' in window) {
    var ob = new MutationObserver(refreshPrices);
    [price, original, savings].forEach(function (n) { ob.observe(n, {childList:true, characterData:true, subtree:true}); });
  }
  var offer = root.querySelector('#cda-offer');
  var dock = root.querySelector('#cda-dock');
  function updateDock() {
    try { dock.hidden = !(offer.getBoundingClientRect().bottom < 0); } catch (e) {}
  }
  window.addEventListener('scroll', updateDock, {passive:true});
  window.addEventListener('resize', updateDock, {passive:true});
  window.addEventListener('load', updateDock);
  window.addEventListener('pageshow', updateDock);
  setInterval(updateDock, 400);
  updateDock();
}());
</script>
